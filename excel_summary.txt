=== Codigo_VBA_y_Manual ===
['GUÍA DE USO Y CÓDIGO MACRO VBA REGISTRO DE ATLETAS']
['PASO 1: HABILITAR MACROS EN EXCEL']
["1. Guarde este archivo con la extensión '.xlsm' (Libro de Excel habilitado para macros)."]
["2. Al abrir el archivo por primera vez, haga clic en 'Habilitar contenido' o 'Habilitar edición' en la barra superior amarilla."]
['PASO 2: INSTALAR EL CÓDIGO MACRO']
["1. Presione las teclas 'ALT + F11' en su teclado para abrir el Editor de Visual Basic (VBA)."]
["2. En el menú superior de VBA, haga clic en 'Insertar' -> 'Módulo'."]
['3. Copie todo el código VBA que se muestra abajo y péguelo en la ventana del Módulo.']
['4. Cierre la ventana de Visual Basic (ALT + Q) y guarde el archivo.']
['PASO 3: ORGANIZACIÓN DE LAS FOTOS DE LOS ATLETAS']
["1. Cree una carpeta llamada 'fotos' en la misma ubicación donde guardó este archivo de Excel."]
['2. Coloque allí las imágenes tipo carnet de cada atleta en formato JPG o PNG.']
["3. En la columna 'Ruta / Nombre Foto' de la hoja 'Base_de_Datos', escriba el nombre exacto del archivo (ejemplo: 'carlos_mendoza.jpg')."]
['CÓDIGO MACRO VBA PARA COPIAR Y PEGAR (MÓDULO DE EXCEL)']
['\' ==============================================================================\n\' CÓDIGO MACRO VBA: CONTROL DE ATLETAS, CARGA DE FOTOS Y GENERACIÓN DE PDFs\n\' ==============================================================================\n\nSub CargarFotoAtleta()\n    \' Carga la foto del atleta seleccionado en la Ficha Individual\n    Dim wsInd As Worksheet\n    Dim wsDB As Worksheet\n    Dim nombreFoto As String\n    Dim rutaCarpeta As String\n    Dim rutaCompleta As String\n    Dim shp As Shape\n    Dim rangoFoto As Range\n    \n    Set wsInd = ThisWorkbook.Sheets("Ficha_Individual")\n    Set wsDB = ThisWorkbook.Sheets("Base_de_Datos")\n    \n    \' Obtener nombre de la foto cargada en Ficha Individual (Celda C24)\n    nombreFoto = wsInd.Range("C24").Value\n    \n    \' Definir rango de la celda donde va la foto (B6:C12)\n    Set rangoFoto = wsInd.Range("B6:C12")\n    \n    \' Eliminar fotos previas en el rango\n    For Each shp In wsInd.Shapes\n        If Not shp.TopLeftCell Is Nothing Then\n            If Not Intersect(shp.TopLeftCell, rangoFoto) Is Nothing Then\n                shp.Delete\n            End If\n        End If\n    Next shp\n    \n    If Trim(nombreFoto) = "" Or nombreFoto = " " Then\n        MsgBox "No hay una foto asignada para este atleta.", vbInformation, "Atención"\n        Exit Sub\n    End If\n    \n    \' Ruta de la carpeta de fotos (Carpeta \'fotos\' en la misma ruta del libro)\n    rutaCarpeta = ThisWorkbook.Path & "\\fotos\\"\n    rutaCompleta = rutaCarpeta & nombreFoto\n    \n    \' Verificar si existe el archivo de imagen\n    If Dir(rutaCompleta) <> "" Then\n        Dim img As Picture\n        Set img = wsInd.Pictures.Insert(rutaCompleta)\n        With img\n            .Left = rangoFoto.Left + 2\n            .Top = rangoFoto.Top + 2\n            .Width = rangoFoto.Width - 4\n            .Height = rangoFoto.Height - 4\n            .Placement = xlMoveAndSize\n        End With\n    Else\n        MsgBox "No se encontró el archivo de foto en: " & rutaCompleta & vbCrLf & _\n               "Asegúrese de guardar la imagen en la carpeta \'fotos\'.", vbWarning, "Foto No Encontrada"\n    End If\nEnd Sub\n\nSub ExportarFichaPDF()\n    \' Exporta la Ficha Individual actual a un archivo PDF\n    Dim wsInd As Worksheet\n    Dim idAtleta As String\n    Dim rutaPDF As String\n    \n    Set wsInd = ThisWorkbook.Sheets("Ficha_Individual")\n    idAtleta = wsInd.Range("C4").Value\n    \n    If idAtleta = "" Then\n        MsgBox "Por favor seleccione un ID de atleta válido.", vbExclamation, "Error"\n        Exit Sub\n    End If\n    \n    \' Generar o actualizar foto antes de exportar\n    Call CargarFotoAtleta\n    \n    rutaPDF = ThisWorkbook.Path & "\\Ficha_Atleta_" & idAtleta & ".pdf"\n    \n    wsInd.ExportAsFixedFormat Type:=xlTypePDF, Filename:=rutaPDF, _\n        Quality:=xlQualityStandard, IncludeDocProperties:=True, _\n        IgnorePrintAreas:=False, OpenAfterPublish:=True\n        \n    MsgBox "La ficha técnica ha sido exportada con éxito en:" & vbCrLf & rutaPDF, vbInformation, "Éxito"\nEnd Sub\n\nSub ImprimirFichasTodos()\n    \' Imprime o exporta a PDF la lista completa de fichas de la hoja Fichas_Todos_Impresion\n    Dim wsAll As Worksheet\n    Dim rutaPDF As String\n    \n    Set wsAll = ThisWorkbook.Sheets("Fichas_Todos_Impresion")\n    rutaPDF = ThisWorkbook.Path & "\\Catálogo_Fichas_Todos_Atletas.pdf"\n    \n    wsAll.ExportAsFixedFormat Type:=xlTypePDF, Filename:=rutaPDF, _\n        Quality:=xlQualityStandard, IncludeDocProperties:=True, _\n        IgnorePrintAreas:=False, OpenAfterPublish:=True\n        \n    MsgBox "El catálogo completo de fichas se exportó a PDF:" & vbCrLf & rutaPDF, vbInformation, "Exportación Completada"\nEnd Sub\n']
